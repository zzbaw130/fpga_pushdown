% CPU architecture: application owns the graph; library schedules dependencies.
\begin{tikzpicture}[font=\small, >=Latex,
 box/.style={draw, rounded corners=2pt, align=center, minimum height=10mm},
 host/.style={box, fill=blue!6}, device/.style={box, fill=orange!9},
 flow/.style={->, thick}, hint/.style={->, dashed, thick, teal!70!black}]
\node[host,text width=38mm] (planner) at (0,3.0) {PostgreSQL planner\\node-level HOST/CSD choices};
\node[host,text width=38mm] (app) at (0,1.4) {Application orchestrator\\operators, buffers, dependencies};
\node[host,text width=38mm] (snia) at (0,-.2) {SNIA library + FEMU plugin\\handles, batches, NVMe mapping};
\node[host,text width=38mm] (cache) at (0,-2.0) {Linux heap inode cache\\range worker + ordinary reads};
\node[device,text width=42mm] (cu) at (6,2.3) {Device CPU compute units\\Scan / Filter / Join / Agg / Sort};
\node[device,text width=42mm] (memory) at (6,.4) {AFDM: input / output batches\\shared private work memory};
\node[device,text width=42mm] (nvm) at (6,-1.5) {NVM namespace\\heap / index / spill extents};
\node[device,text width=28mm] (log) at (10,-1.5) {AER + range log\\completed scan hints};
\draw[flow] (planner) -- (app);
\draw[flow] (app) -- (snia);
\draw[flow] (snia.east) -- node[above,sloped,font=\scriptsize]{individual I/O / compute requests} (cu.west);
\draw[flow, <->] (cu) -- (memory);
\draw[flow, <->] (memory) -- node[right,font=\scriptsize]{internal LOAD / STORE} (nvm);
\draw[flow] (cu.north) -- ++(0,.8) -| node[pos=.3,above,font=\scriptsize]{completion events / final result batches} (app.north east);
\draw[hint] (cu.east) -| (log.north);
\draw[hint] (log.south) -- ++(0,-1.2) -| node[pos=.28,below,font=\scriptsize]{notice then fetch range metadata} (cache.south);
\draw[hint] (cache.east) -- node[below,sloped,font=\scriptsize]{refill through ordinary inode reads} (nvm.west);
\draw[hint] (cache.west) -- ++(-.9,0) |- node[pos=.7,left,font=\scriptsize]{observed residency} (planner.west);
\node[draw,dashed,fit=(planner)(app)(snia)(cache),inner sep=5mm,label=above:{Host application and operating system}] {};
\node[draw,dashed,fit=(cu)(memory)(nvm)(log),inner sep=5mm,label=above:{CPU-based computational storage device}] {};
\end{tikzpicture}
